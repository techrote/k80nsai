cmake_minimum_required(VERSION 3.19)
project(k80nsai_q1_host_tests LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../reference/q1" q1-reference)
add_executable(k05-reference-tests reference_tests.cpp)
target_link_libraries(k05-reference-tests PRIVATE k80nsai-q1-reference)
add_executable(k05-native-abi native_abi.cpp)
target_compile_features(k05-native-abi PRIVATE cxx_std_17)
target_include_directories(k05-native-abi PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../vendor/llama.cpp/ggml/src")
add_test(NAME k05-native-abi COMMAND k05-native-abi)
add_test(NAME k05-reference COMMAND k05-reference-tests)
add_test(NAME k05-negative-controls COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/negative_controls.py" "$<TARGET_FILE:k05-reference-tests>")
set_tests_properties(k05-native-abi k05-reference k05-negative-controls PROPERTIES TIMEOUT 60)
